-- [[ UNC Luau - Chức năng Bất tử (Godmode) ]]
local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local CoreGui = game:GetService("CoreGui")

local player = Players.LocalPlayer

-- Xóa GUI cũ nếu đã tồn tại để tránh trùng lặp
if CoreGui:FindFirstChild("GodmodeGUI") then
    CoreGui.GodmodeGUI:Destroy()
end

-- Tạo GUI chính
local ScreenGui = Instance.new("ScreenGui")
ScreenGui.Name = "GodmodeGUI"
ScreenGui.Parent = CoreGui
ScreenGui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling

local MainFrame = Instance.new("Frame")
MainFrame.Name = "MainFrame"
MainFrame.Parent = ScreenGui
MainFrame.BackgroundColor3 = Color3.fromRGB(25, 25, 35)
MainFrame.Position = UDim2.new(0.5, -110, 0.5, -75)
MainFrame.Size = UDim2.new(0, 220, 0, 140)
MainFrame.Active = true
MainFrame.Draggable = true

local UICorner = Instance.new("UICorner")
UICorner.CornerRadius = UDim.new(0, 10)
UICorner.Parent = MainFrame

local UIStroke = Instance.new("UIStroke")
UIStroke.Color = Color3.fromRGB(80, 150, 255)
UIStroke.Thickness = 2
UIStroke.Parent = MainFrame

local Title = Instance.new("TextLabel")
Title.Name = "Title"
Title.Parent = MainFrame
Title.BackgroundTransparency = 1.00
Title.Position = UDim2.new(0, 0, 0, 10)
Title.Size = UDim2.new(1, 0, 0, 30)
Title.Font = Enum.Font.GothamBold
Title.Text = "SCRIPT BẤT TỬ"
Title.TextColor3 = Color3.fromRGB(255, 255, 255)
Title.TextSize = 16

local ToggleBtn = Instance.new("TextButton")
ToggleBtn.Name = "ToggleBtn"
ToggleBtn.Parent = MainFrame
ToggleBtn.BackgroundColor3 = Color3.fromRGB(200, 50, 50)
ToggleBtn.Position = UDim2.new(0.1, 0, 0.45, 0)
ToggleBtn.Size = UDim2.new(0.8, 0, 0, 45)
ToggleBtn.Font = Enum.Font.GothamBold
ToggleBtn.Text = "Trạng thái: TẮT"
ToggleBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
ToggleBtn.TextSize = 14

local BtnCorner = Instance.new("UICorner")
BtnCorner.CornerRadius = UDim.new(0, 8)
BtnCorner.Parent = ToggleBtn

-- Logic Bất tử
local godmodeEnabled = false
local connection = nil

local function enableGodmode()
    local char = player.Character or player.CharacterAdded:Wait()
    local humanoid = char:WaitForChild("Humanoid")
    
    humanoid.MaxHealth = math.huge
    humanoid.Health = math.huge
    
    if connection then
        connection:Disconnect()
    end
    
    connection = RunService.Heartbeat:Connect(function()
        if godmodeEnabled and char and humanoid and humanoid.Parent then
            if humanoid.Health < humanoid.MaxHealth then
                humanoid.Health = humanoid.MaxHealth
            end
            pcall(function()
                humanoid:SetStateEnabled(Enum.HumanoidStateType.Dead, false)
            end)
        end
    end)
end

local function disableGodmode()
    if connection then
        connection:Disconnect()
        connection = nil
    end
    local char = player.Character
    if char then
        local humanoid = char:FindFirstChildOfClass("Humanoid")
        if humanoid then
            humanoid.MaxHealth = 100
            humanoid.Health = 100
            pcall(function()
                humanoid:SetStateEnabled(Enum.HumanoidStateType.Dead, true)
            end)
        end
    end
end

ToggleBtn.MouseButton1Click:Connect(function()
    godmodeEnabled = not godmodeEnabled
    if godmodeEnabled then
        ToggleBtn.Text = "Trạng thái: BẬT"
        ToggleBtn.BackgroundColor3 = Color3.fromRGB(50, 200, 50)
        enableGodmode()
    else
        ToggleBtn.Text = "Trạng thái: TẮT"
        ToggleBtn.BackgroundColor3 = Color3.fromRGB(200, 50, 50)
        disableGodmode()
    end
end)

player.CharacterAdded:Connect(function(newChar)
    if godmodeEnabled then
        task.wait(1)
        enableGodmode()
    end
end)
